### Written by Oldbird
### These are sets of triggers meant to be used to specifiy ideologies in triggers.




## The following triggers are specifically for the country leader and ->
## are based out of ROOT and then party_leader.
## Some trigger has a small comment to see exactly why and what they do
## But the names themselves are descriptive

# First checks if the nation has elections, then goes through any sub_ideologies that can have
# democratic tendencies. Communism and nationalism are included because of internal elections in
# states still happening. But the has_elections trigger will still remove the non-democratic nations
has_democratic_leader = {
	ROOT = {
		has_elections = yes
		party_leader = {
			limit = {
				OR = {
					has_green_leader = yes
					has_centrist_leader = yes
					has_conservative_leader = yes
					has_socialist_leader = yes
					has_social_democrat_leader = yes
					has_communist_leader = yes
				}
			}
		}
	}
}
